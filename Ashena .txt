  <div id="basket" class="hidden">🧺</div>

  <!-- صفحه شروع -->
  <div id="overlay">
    <h1>🧺 سبد و میوه 🍎</h1>
    <p>اول باید بگی آشنات کیه...</p>
    <input id="nameInput" type="text" placeholder="آشنات کی هست؟" />
    <button id="checkBtn">بگو ببینم</button>
    <div id="msg" class="msg"></div>
  </div>

<script>
(function () {
  const game = document.getElementById('game');
  const overlay = document.getElementById('overlay');
  const msg = document.getElementById('msg');
  const nameInput = document.getElementById('nameInput');
  const checkBtn = document.getElementById('checkBtn');
  const hud = document.getElementById('hud');
  const basket = document.getElementById('basket');
  const scoreEl = document.getElementById('score');
  const livesEl = document.getElementById('lives');

  let running = false;
  let score = 0;
  let lives = 3;
  let items = [];
  let spawnTimer = null;
  let rafId = null;
  let lastTime = 0;
  let spawnInterval = 900;
  let fallSpeedBase = 0.12;

  const GAME_W = () => game.clientWidth;
  const GAME_H = () => game.clientHeight;
  const BASKET_W = 70;

  let basketX = 0;

  function setBasketX(x) {
    const max = GAME_W() - BASKET_W;
    basketX = Math.max(0, Math.min(max, x));
    basket.style.left = basketX + 'px';
  }

  function handleMove(clientX) {
    const rect = game.getBoundingClientRect();
    const x = clientX - rect.left - BASKET_W / 2;
    setBasketX(x);
  }

  game.addEventListener('mousemove', e => { if (running) handleMove(e.clientX); });
  game.addEventListener('touchmove', e => {
    if (!running) return;
    e.preventDefault();
    handleMove(e.touches[0].clientX);
  }, { passive: false });
  game.addEventListener('touchstart', e => {
    if (!running) return;
    handleMove(e.touches[0].clientX);
  });

  checkBtn.addEventListener('click', checkName);
  nameInput.addEventListener('keydown', e => {
    if (e.key === 'Enter') checkName();
  });

  function checkName() {
    const val = nameInput.value.trim();
    if (!val) {
      msg.className = 'msg yellow';
      msg.textContent = 'یه اسمی بنویس 🙂';
      return;
    }

    if (!val.includes('کمالپور')) {
      msg.className = 'msg red';
      msg.textContent = 'شرمنده، این آشنا رو نمی‌شناسیم! فقط آشناهای کمالپور می‌تونن بازی کنن 🙏';
      return;
    }

    // مرحله ۱
    msg.className = 'msg green';
    msg.textContent = 'بله بله، آشناست! از آشناهای ما هستن 😄';

    checkBtn.disabled = true;
    nameInput.disabled = true;

    // مرحله ۲
    setTimeout(() => {
      msg.className = 'msg yellow';
      msg.textContent = 'اره، خیلی‌ها ازش می‌ترسن! 😅';
    }, 1500);

    // شروع بازی
    setTimeout(() => {
      startGame();
    }, 3200);
  }

  function startGame() {
    overlay.classList.add('hidden');
    hud.classList.remove('hidden');
    basket.classList.remove('hidden');

    score = 0;
    lives = 3;
    spawnInterval = 900;
    fallSpeedBase = 0.12;
    scoreEl.textContent = '0';
    livesEl.textContent = '3';

    items.forEach(it => it.el.remove());
    items = [];

    setBasketX(GAME_W() / 2 - BASKET_W / 2);

    running = true;
    lastTime = performance.now();
    clearInterval(spawnTimer);
    spawnTimer = setInterval(spawnItem, spawnInterval);
    rafId = requestAnimationFrame(loop);
  }

  const FRUITS = ['🍎','🍊','🍋','🍇','🍉','🍓','🍒','🥝','🍑','🍍'];
  const BOMB = '💣';

  function spawnItem() {
    if (!running) return;

    const level = Math.floor(score / 10);
    spawnInterval = Math.max(350, 900 - level * 60);
    fallSpeedBase = 0.12 + level * 0.02;

    const isBomb = Math.random() < 0.22;
    const el = document.createElement('div');
    el.className = 'item';
    el.textContent = isBomb ? BOMB : FRUITS[Math.floor(Math.random() * FRUITS.length)];

    const size = 40;
    const x = Math.random() * (GAME_W() - size);
    el.style.left = x + 'px';
    el.style.top = '-50px';
    game.appendChild(el);

    items.push({
      el, x, y: -50, size, isBomb,
      speed: fallSpeedBase * (0.8 + Math.random() * 0.5)
    });
  }

  function loop(now) {
    if (!running) return;
    const dt = Math.min(50, now - lastTime);
    lastTime = now;

    const basketY = GAME_H() - 70;

    for (let i = items.length - 1; i >= 0; i--) {
      const it = items[i];
      it.y += it.speed * dt;
      it.el.style.top = it.y + 'px';

      const hitX = it.x + it.size > basketX + 8 && it.x < basketX + BASKET_W - 8;
      const hitY = it.y + it.size > basketY && it.y < basketY + 55;

      if (hitX && hitY) {
        if (it.isBomb) {
          lives--;
          livesEl.textContent = lives;
          if (lives <= 0) { gameOver(); return; }
        } else {
          score++;
          scoreEl.textContent = score;
        }
        it.el.remove();
        items.splice(i, 1);
        continue;
      }

      if (it.y > GAME_H()) {
        if (!it.isBomb) {
          lives--;
          livesEl.textContent = lives;
          if (lives <= 0) { gameOver(); return; }
        }
        it.el.remove();
        items.splice(i, 1);
      }
    }

    rafId = requestAnimationFrame(loop);
  }

  function gameOver() {
    running = false;
    clearInterval(spawnTimer);
    cancelAnimationFrame(rafId);

    items.forEach(it => it.el.remove());
    items = [];

    hud.classList.add('hidden');
    basket.classList.add('hidden');

    overlay.classList.remove('hidden');
    overlay.innerHTML = `
      <h1>بازی تموم شد!</h1>
      <p>امتیازت: <b>${score}</b></p>
      <p>${score >= 30 ? 'آفرین! حرفه‌ای بودی 🏆' : score >= 15 ? 'خوب بود! 👏' : 'بار بعد بهتر می‌شی 💪'}</p>
      <button id="againBtn" style="background:#4caf50;">دوباره بازی</button>
    `;
    document.getElementById('againBtn').addEventListener('click', () => {
      location.reload();
    });
  }
})();
</script>
